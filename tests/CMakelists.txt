find_package(GTest REQUIRED)

ccp_add_executable(CcpParserTest
    basic.cpp
    functions.cpp
    helpers.cpp
    operators.cpp
    pch.cpp
    variables.cpp
)
target_sources(CcpParserTest PRIVATE
    helpers.h
    pch.h
)

target_precompile_headers(CcpParserTest PRIVATE pch.h)
target_link_libraries(CcpParserTest PRIVATE CcpParser GTest::GTest GTest::Main)
set_target_properties(CcpParserTest PROPERTIES FOLDER "Tests")
gtest_discover_tests(CcpParserTest)

if (WIN32)
    # HACK: Until PLAT-11397 issue is resolved
    add_custom_command(TARGET CcpParserTest PRE_BUILD
        COMMAND ${CMAKE_COMMAND} -E echo "Copying $<TARGET_FILE:GTest::gtest> -> $<TARGET_FILE_DIR:CcpParserTest>"
        COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:GTest::gtest>" "$<TARGET_FILE_DIR:CcpParserTest>"
    )
    add_custom_command(TARGET CcpParserTest PRE_BUILD
        COMMAND ${CMAKE_COMMAND} -E echo "Copying $<TARGET_FILE:GTest::gtest_main> -> $<TARGET_FILE_DIR:CcpParserTest>"
        COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:GTest::gtest_main>" "$<TARGET_FILE_DIR:CcpParserTest>"
    )
endif()
